cmake_minimum_required(VERSION 3.20)
project(ros2_stage2 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(lifecycle_msgs REQUIRED)
add_executable(stage2_node stage2_node.cpp)
ament_target_dependencies(stage2_node rclcpp rclcpp_action rclcpp_lifecycle std_msgs example_interfaces lifecycle_msgs)
install(TARGETS stage2_node RUNTIME DESTINATION bin)
ament_package()
