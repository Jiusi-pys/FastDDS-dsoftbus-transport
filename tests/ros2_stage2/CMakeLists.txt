cmake_minimum_required(VERSION 3.20)
project(ros2_stage2 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(std_msgs REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(statistics_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(type_description_interfaces REQUIRED)
add_executable(stage2_node stage2_node.cpp)
ament_target_dependencies(stage2_node rclcpp rclcpp_action rclcpp_lifecycle std_msgs example_interfaces lifecycle_msgs)
add_executable(stage2_features stage2_features.cpp)
ament_target_dependencies(stage2_features rclcpp rclcpp_action std_msgs example_interfaces geometry_msgs rcl_interfaces
  rosgraph_msgs sensor_msgs statistics_msgs tf2_ros type_description_interfaces)
install(TARGETS stage2_node stage2_features RUNTIME DESTINATION bin)
ament_package()
