cmake_minimum_required(VERSION 3.22)
project(DirectChildSupervision LANGUAGES C CXX)
enable_testing()
add_library(owned_policy OBJECT owned_policy.c)
target_include_directories(owned_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(owned_policy PUBLIC c_std_11)
set_target_properties(owned_policy PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(WIN32)
  add_executable(supervision_finite_child finite_child.cpp)
  add_executable(supervision_policy_test policy_test.cpp)
  target_link_libraries(supervision_policy_test PRIVATE owned_policy)
  foreach(t owned_policy supervision_finite_child supervision_policy_test)
    target_compile_options(${t} PRIVATE /W4 /WX /MT)
    set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
  endforeach()
  target_compile_options(supervision_policy_test PRIVATE /EHsc)
  target_compile_features(supervision_policy_test PRIVATE cxx_std_17)
  target_compile_features(supervision_finite_child PRIVATE cxx_std_17)
  add_test(NAME stage1.supervision.direct_child COMMAND supervision_policy_test "$<TARGET_FILE:supervision_finite_child>")
  set_tests_properties(stage1.supervision.direct_child PROPERTIES TIMEOUT 25)
elseif(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_executable(owned_supervisor_direct supervisor_native.c)
  target_link_libraries(owned_supervisor_direct PRIVATE owned_policy)
  foreach(t owned_policy owned_supervisor_direct)
    target_compile_options(${t} PRIVATE -O2 -Wall -Wextra -Werror -fPIE)
  endforeach()
  target_link_options(owned_supervisor_direct PRIVATE "LINKER:-z,defs" "LINKER:--no-undefined" "LINKER:-z,now")
else()
  message(FATAL_ERROR "Only explicit Windows Host or OHOS crossbuild")
endif()
