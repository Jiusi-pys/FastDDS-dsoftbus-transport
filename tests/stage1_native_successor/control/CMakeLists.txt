cmake_minimum_required(VERSION 3.22)
project(Stage1ControlIO LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Real Host fixture driver is Windows-only; Native frontend uses the parent explicit OHOS target")
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(control_fixture SHARED fixture.cpp)
add_executable(control_io_test test_control.cpp wire.cpp io.cpp v2_client.cpp frontend_control.cpp native_ingress.cpp deployment.cpp parent_protocol.cpp v2_route.cpp)
target_compile_definitions(control_io_test PRIVATE STAGE1_CONTROL_HOST_FIXTURE=1)
foreach(t control_fixture control_io_test)
  target_compile_features(${t} PRIVATE cxx_std_17)
  target_compile_options(${t} PRIVATE /W4 /WX /EHsc /MD)
endforeach()
# The evidence runner supplies actual EXE/deployment hashes to the test command;
# neither field grants permission or turns fixture evidence into native proof.
add_test(NAME stage1.controlio.contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_case.py" "$<TARGET_FILE:control_io_test>" "$<TARGET_FILE:control_fixture>")
set_tests_properties(stage1.controlio.contract PROPERTIES TIMEOUT 25)
add_executable(control_handoff_test handoff_test.cpp handoff_win.cpp deployment.cpp wire.cpp io.cpp v2_client.cpp frontend_control.cpp native_ingress.cpp parent_protocol.cpp v2_route.cpp)
target_compile_features(control_handoff_test PRIVATE cxx_std_17)
target_compile_definitions(control_handoff_test PRIVATE STAGE1_CONTROL_HOST_FIXTURE=1)
target_compile_options(control_handoff_test PRIVATE /W4 /WX /EHsc /MT)
set_target_properties(control_handoff_test PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
add_test(NAME stage1.handoff.contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_handoff.py" "$<TARGET_FILE:control_handoff_test>" "$<TARGET_FILE:handoff_fixture_noexcept>" "${CMAKE_CURRENT_BINARY_DIR}/handoff-results")
set_tests_properties(stage1.handoff.contract PROPERTIES TIMEOUT 25)
add_library(handoff_fixture_noexcept SHARED fixture_noexcept.cpp)
target_compile_features(handoff_fixture_noexcept PRIVATE cxx_std_17)
target_compile_options(handoff_fixture_noexcept PRIVATE /W4 /WX /EHsc /MD)
# Evidence build seals this target's SHA and passes only that immutable value as
# STAGE1_BOUND_FIXTURE_SHA when compiling control_handoff_test. No runtime flag
# or manifest value can authorize a Native loader or select a different DLL.
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/handoff_fixture_bound.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/write_fixture_hash.py" "$<TARGET_FILE:handoff_fixture_noexcept>" "${CMAKE_CURRENT_BINARY_DIR}/handoff_fixture_bound.h"
  DEPENDS handoff_fixture_noexcept "${CMAKE_CURRENT_SOURCE_DIR}/write_fixture_hash.py" VERBATIM)
add_custom_target(handoff_fixture_binding DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/handoff_fixture_bound.h")
add_dependencies(control_handoff_test handoff_fixture_binding)
set_source_files_properties(handoff_test.cpp handoff_win.cpp PROPERTIES COMPILE_OPTIONS "/FI${CMAKE_CURRENT_BINARY_DIR}/handoff_fixture_bound.h")
add_executable(native_preparation_contract test_native_contract.cpp native_ingress.cpp deployment.cpp parent_protocol.cpp)
target_compile_features(native_preparation_contract PRIVATE cxx_std_17)
target_compile_options(native_preparation_contract PRIVATE /W4 /WX /EHsc /MT)
set_target_properties(native_preparation_contract PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
add_test(NAME stage1.native_preparation.contract COMMAND native_preparation_contract)
set_tests_properties(stage1.native_preparation.contract PROPERTIES TIMEOUT 25)
add_executable(parent_v2_contract test_parent_v2.cpp parent_protocol.cpp native_ingress.cpp deployment.cpp v2_route.cpp io.cpp wire.cpp v2_client.cpp)
target_compile_features(parent_v2_contract PRIVATE cxx_std_17)
target_compile_options(parent_v2_contract PRIVATE /W4 /WX /EHsc /MT)
set_target_properties(parent_v2_contract PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreaded)
add_test(NAME stage1.parent_v2.contract COMMAND parent_v2_contract)
set_tests_properties(stage1.parent_v2.contract PROPERTIES TIMEOUT 25)
