cmake_minimum_required(VERSION 3.22)
project(Stage1NativeSuccessor LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR NOT OHOS_STL STREQUAL "c++_shared")
    message(FATAL_ERROR "Explicit OHOS ARM64/shared-SDK toolchain required")
endif()
foreach(p IN ITEMS PLATFORM_CXX_HEADERS PLATFORM_CPP_LIBRARY PLATFORM_TOKEN_LIBRARY PLATFORM_ACCESS_LIBRARY DEPLOYED_LIBC_LIBRARY)
    if(NOT IS_ABSOLUTE "${${p}}" OR NOT EXISTS "${${p}}")
        message(FATAL_ERROR "Missing explicit existing ABI/CFI input ${p}")
    endif()
endforeach()
find_package(FastDDSDSoftBusStock CONFIG REQUIRED)
find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
find_package(fastcdr 2.2.5 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(stage1_native_frontend frontend.cpp entry_admission.cpp layout_c.c control/wire.cpp control/io.cpp control/v2_client.cpp control/frontend_control.cpp control/deployment.cpp control/native_ingress.cpp control/parent_protocol.cpp control/v2_route.cpp)
add_executable(stage1_native_parent native_parent_main.cpp control/native_parent.cpp control/native_ingress.cpp control/deployment.cpp control/parent_protocol.cpp control/wire.cpp control/io.cpp)
add_library(stage1_actionbag_identity SHARED identity_bridge.cpp)
add_library(stage1_native_consumer SHARED consumer_module.cpp consumer_run.cpp run_layout.c)
add_library(stage1_native_loader_fixture SHARED loader_fixture.c)
foreach(t IN ITEMS stage1_actionbag_identity stage1_native_consumer stage1_native_loader_fixture)
    target_link_options(${t} PRIVATE "LINKER:-soname,$<TARGET_FILE_NAME:${t}>")
endforeach()
foreach(t IN ITEMS stage1_native_frontend stage1_native_parent stage1_actionbag_identity stage1_native_consumer stage1_native_loader_fixture)
    set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE ON BUILD_RPATH "" INSTALL_RPATH "")
    target_compile_options(${t} PRIVATE -O2 -Wall -Wextra -Werror -Wno-error=unused-command-line-argument -H -flto -fsanitize=cfi -fsanitize-cfi-cross-dso -fvisibility=hidden)
    target_link_options(${t} PRIVATE -flto -fsanitize=cfi -fsanitize-cfi-cross-dso "LINKER:-z,defs" "LINKER:-z,now" "LINKER:--no-undefined" "LINKER:--no-allow-shlib-undefined" "LINKER:-Map,${CMAKE_CURRENT_BINARY_DIR}/${t}.map")
    target_link_libraries(${t} PRIVATE "${DEPLOYED_LIBC_LIBRARY}")
    foreach(dir IN LISTS NATIVE_PROVIDER_DIRS)
        if(NOT IS_ABSOLUTE "${dir}" OR NOT IS_DIRECTORY "${dir}")
            message(FATAL_ERROR "Missing declared provider directory")
        endif()
        target_link_options(${t} PRIVATE "LINKER:-rpath-link,${dir}")
    endforeach()
endforeach()
target_link_libraries(stage1_native_frontend PRIVATE Threads::Threads dl)
target_link_libraries(stage1_native_parent PRIVATE Threads::Threads dl)
# Keep the qualified SDK C++ executable-root runtime explicit. This is not a
# namespace alias or C-only loader qualification.
target_link_options(stage1_native_frontend PRIVATE "LINKER:--no-as-needed" -lc++_shared "LINKER:--as-needed")
target_link_options(stage1_native_parent PRIVATE "LINKER:--no-as-needed" -lc++_shared "LINKER:--as-needed")
target_compile_options(stage1_actionbag_identity PRIVATE -nostdinc++ -isystem "${PLATFORM_CXX_HEADERS}")
target_include_directories(stage1_actionbag_identity PRIVATE "${ACCESS_TOKEN_HEADERS}" "${TOKEN_SETPROC_HEADERS}")
target_link_options(stage1_actionbag_identity PRIVATE -nostdlib++)
target_link_libraries(stage1_actionbag_identity PRIVATE "${PLATFORM_CPP_LIBRARY}" "${PLATFORM_TOKEN_LIBRARY}" "${PLATFORM_ACCESS_LIBRARY}")
target_link_libraries(stage1_native_consumer PRIVATE FastDDSDSoftBusStock::Transport Threads::Threads)
set(CMAKE_SKIP_RPATH TRUE)
