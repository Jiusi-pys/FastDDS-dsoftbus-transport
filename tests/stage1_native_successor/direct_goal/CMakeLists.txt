cmake_minimum_required(VERSION 3.22)
project(Stage1DirectGoal LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS" OR NOT OHOS_STL STREQUAL "c++_shared")
  message(FATAL_ERROR "Explicit existing SDK23 OHOS ARM64/shared profile required")
endif()
foreach(p IN ITEMS GOAL_SCALAR_LIBRARY GOAL_SOFTBUS_LIBRARY GOAL_IDENTITY_BRIDGE DEPLOYED_LIBC_LIBRARY)
  if(NOT IS_ABSOLUTE "${${p}}" OR NOT EXISTS "${${p}}")
    message(FATAL_ERROR "Missing exact existing native input ${p}")
  endif()
endforeach()
find_package(FastDDSDSoftBusStock CONFIG REQUIRED)
find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
find_package(fastcdr 2.2.5 EXACT CONFIG REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(SHA256 "${GOAL_SCALAR_LIBRARY}" scalar)
file(SHA256 "${GOAL_SOFTBUS_LIBRARY}" softbus)
file(SHA256 "${GOAL_IDENTITY_BRIDGE}" bridge)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/goal_base.h" "#pragma once\n#define GOAL_SCALAR_SHA \"${scalar}\"\n#define GOAL_SOFTBUS_SHA \"${softbus}\"\n#define GOAL_BRIDGE_SHA \"${bridge}\"\n")
add_executable(own_goal_runner direct_main.cpp native_files.cpp goal_control.cpp ../entry_admission.cpp ../consumer_run.cpp ../run_layout.c ../control/wire.cpp)
add_executable(own_goal_launcher own_exec.cpp native_files.cpp ../entry_admission.cpp)
add_executable(goal_supervisor goal_supervisor.c goal_policy.c)
set_target_properties(goal_supervisor PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE ON)
target_compile_options(goal_supervisor PRIVATE -O2 -Wall -Wextra -Werror -flto -fsanitize=cfi -fsanitize-cfi-cross-dso -fvisibility=hidden)
target_link_options(goal_supervisor PRIVATE -flto -fsanitize=cfi -fsanitize-cfi-cross-dso "LINKER:-z,defs" "LINKER:-z,now" "LINKER:--no-undefined" "LINKER:--no-allow-shlib-undefined")
target_link_libraries(goal_supervisor PRIVATE "${DEPLOYED_LIBC_LIBRARY}")
foreach(t own_goal_runner own_goal_launcher)
  set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE ON BUILD_RPATH "" INSTALL_RPATH "")
  target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_compile_options(${t} PRIVATE -O2 -Wall -Wextra -Werror -Wno-error=unused-command-line-argument -H -flto -fsanitize=cfi -fsanitize-cfi-cross-dso -fvisibility=hidden)
  target_link_options(${t} PRIVATE -flto -fsanitize=cfi -fsanitize-cfi-cross-dso "LINKER:-z,defs" "LINKER:-z,now" "LINKER:--no-undefined" "LINKER:--no-allow-shlib-undefined" "LINKER:-Map,${CMAKE_CURRENT_BINARY_DIR}/${t}.map" "LINKER:--no-as-needed" -lc++_shared "LINKER:--as-needed")
  target_link_libraries(${t} PRIVATE "${DEPLOYED_LIBC_LIBRARY}" dl pthread)
  foreach(d IN LISTS NATIVE_PROVIDER_DIRS)
    target_link_options(${t} PRIVATE "LINKER:-rpath-link,${d}")
  endforeach()
endforeach()
target_include_directories(own_goal_runner PRIVATE "${SOFTBUS_BUS_HEADER_DIR}" "${SOFTBUS_COMMON_HEADER_DIR}")
target_link_options(own_goal_launcher PRIVATE "LINKER:-rpath,$ORIGIN/lib")
target_link_libraries(own_goal_runner PRIVATE FastDDSDSoftBusStock::Transport)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/goal_bound.h"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/write_app_binding.py" "$<TARGET_FILE:own_goal_runner>" "${CMAKE_CURRENT_BINARY_DIR}/goal_bound.h"
  DEPENDS own_goal_runner "${CMAKE_CURRENT_SOURCE_DIR}/write_app_binding.py" VERBATIM)
add_custom_target(goal_binding DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/goal_bound.h")
add_dependencies(own_goal_launcher goal_binding)
set(CMAKE_SKIP_RPATH TRUE)
