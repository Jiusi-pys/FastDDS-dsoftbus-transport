cmake_minimum_required(VERSION 3.22)
project(Stage1DelayedHost LANGUAGES C CXX)
if(NOT WIN32 OR CMAKE_CROSSCOMPILING OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This target qualifies only Windows x64 Host fixtures")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_executable(stage1_delayed_frontend frontend.c abi_cpp.cpp)
set_target_properties(stage1_delayed_frontend PROPERTIES ENABLE_EXPORTS TRUE)
target_compile_definitions(stage1_delayed_frontend PRIVATE DD_FRONTEND=1)
add_library(stage1_delayed_fixture MODULE fixture_module.c)
add_library(stage1_delayed_missing MODULE missing_export.c)
target_link_libraries(stage1_delayed_fixture PRIVATE stage1_delayed_frontend)
target_link_libraries(stage1_delayed_missing PRIVATE stage1_delayed_frontend)
foreach(t IN ITEMS stage1_delayed_frontend stage1_delayed_fixture stage1_delayed_missing)
    set_target_properties(${t} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
    target_compile_options(${t} PRIVATE /W4 /WX)
endforeach()
add_custom_target(stage1_delayed_all DEPENDS stage1_delayed_frontend stage1_delayed_fixture stage1_delayed_missing)
add_test(NAME stage1.delayed_module.contract COMMAND "${Python3_EXECUTABLE}" -I -B
    "${CMAKE_CURRENT_LIST_DIR}/test_frontend.py" --exe "$<TARGET_FILE:stage1_delayed_frontend>"
    --module "$<TARGET_FILE:stage1_delayed_fixture>" --missing-export "$<TARGET_FILE:stage1_delayed_missing>"
    --result "${CMAKE_BINARY_DIR}/child_cases.json")
set_tests_properties(stage1.delayed_module.contract PROPERTIES TIMEOUT 25 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
