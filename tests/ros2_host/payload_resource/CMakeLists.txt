cmake_minimum_required(VERSION 3.22)
project(payload_resource_ros_tests LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(FastDDSDSoftBusTransportRos2Host CONFIG REQUIRED)
get_filename_component(_guard_prefix "${FastDDSDSoftBusTransportRos2Host_DIR}/../../.." ABSOLUTE)
add_executable(payload_resource_ros_tests main.cpp)
target_include_directories(payload_resource_ros_tests PRIVATE "${_guard_prefix}/share/FastDDSDSoftBusTransportRos2Host/payload_resource")
ament_target_dependencies(payload_resource_ros_tests rclcpp std_msgs)
target_compile_features(payload_resource_ros_tests PRIVATE cxx_std_17)
target_link_libraries(payload_resource_ros_tests FastDDSDSoftBusTransportRos2Host::factory_configuration)
if(FASTDDS_DSOFTBUS_ROS214_TEST_ASAN)
  fastdds_dsoftbus_test_asan_target(payload_resource_ros_tests)
