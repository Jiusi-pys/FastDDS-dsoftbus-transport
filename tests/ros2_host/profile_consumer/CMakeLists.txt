cmake_minimum_required(VERSION 3.22)
project(ros2_historical_profile_consumer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(FastDDSDSoftBusTransportRos2Host CONFIG REQUIRED)
if(NOT FastDDSDSoftBusTransportRos2Host_PROFILE STREQUAL "historical-2147-security-statistics")
    message(FATAL_ERROR "This qualification consumer requires the explicit historical Host package")
endif()
add_executable(ros2_historical_profile_consumer main.cpp)
ament_target_dependencies(ros2_historical_profile_consumer rclcpp std_msgs example_interfaces)
target_compile_features(ros2_historical_profile_consumer PRIVATE cxx_std_17)
target_link_libraries(ros2_historical_profile_consumer FastDDSDSoftBusTransportRos2Host::factory_configuration)
