cmake_minimum_required(VERSION 3.22)
project(LocatorObservationHostTests LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "This project executes host structural tests, not device tests")
endif()
get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(OBSERVATION_SOURCE_ROOT "${_repo}" CACHE PATH "Source root for the actual observer under test")
option(FASTDDS_DSOFTBUS_LEGACY_OBSERVER "Build the same repetition reproducer against the unchanged failed observer" OFF)
find_package(Threads REQUIRED)
add_executable(locator_observation_tests ../locator_observation_tests.cpp
    "${OBSERVATION_SOURCE_ROOT}/integrations/ros2/target/NativeObservation.cpp"
    "${OBSERVATION_SOURCE_ROOT}/probes/PublicSocketRtpsObservation.cpp")
target_include_directories(locator_observation_tests PRIVATE "${OBSERVATION_SOURCE_ROOT}/integrations/ros2/target")
target_compile_features(locator_observation_tests PRIVATE cxx_std_17)
target_compile_definitions(locator_observation_tests PRIVATE FASTDDS_DSOFTBUS_OBSERVATION_HOST_TEST=1)
if(FASTDDS_DSOFTBUS_LEGACY_OBSERVER)
    target_compile_definitions(locator_observation_tests PRIVATE FASTDDS_DSOFTBUS_LEGACY_OBSERVER=1)
endif()
if(MSVC)
    target_compile_options(locator_observation_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(locator_observation_tests PRIVATE -UNDEBUG)
endif()
target_link_libraries(locator_observation_tests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME locator_observation.repetition COMMAND locator_observation_tests repetition)
if(NOT FASTDDS_DSOFTBUS_LEGACY_OBSERVER)
    foreach(_case keys coherence distinct_capacity counters no_allocation concurrency
                  invalid_frame unsupported_frame parser_capacity invalid_selection selection_capacity advert_capacity)
        add_test(NAME locator_observation.${_case} COMMAND locator_observation_tests ${_case})
    endforeach()
endif()
set_tests_properties(locator_observation.repetition PROPERTIES TIMEOUT 20)
