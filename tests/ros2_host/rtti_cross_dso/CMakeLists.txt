cmake_minimum_required(VERSION 3.22)
project(topic_rtti_cross_dso LANGUAGES CXX)
find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
find_package(fastcdr 2 CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(topic_rtti_producer SHARED producer.cpp)
target_link_libraries(topic_rtti_producer PRIVATE fastrtps fastcdr)
target_compile_definitions(topic_rtti_producer PRIVATE RTTI_PRODUCER=1)
add_executable(topic_rtti_consumer consumer.cpp)
target_link_libraries(topic_rtti_consumer PRIVATE topic_rtti_producer fastrtps)
if(MSVC)
  target_compile_options(topic_rtti_producer PRIVATE /UNDEBUG)
  target_compile_options(topic_rtti_consumer PRIVATE /UNDEBUG)
endif()
enable_testing()
add_test(NAME actual_topic_cross_dso_rtti COMMAND topic_rtti_consumer)
set_tests_properties(actual_topic_cross_dso_rtti PROPERTIES TIMEOUT 20)
