cmake_minimum_required(VERSION 3.20)
project(ros_diagnostic_stream CXX)
find_package(fastrtps 2.14 CONFIG REQUIRED)
find_package(fastcdr 2.2 CONFIG REQUIRED)
add_executable(ros_diagnostic_stream main.cpp ../../../integrations/ros2/FactoryAbiProbe.cpp)
target_compile_features(ros_diagnostic_stream PRIVATE cxx_std_17)
target_compile_definitions(ros_diagnostic_stream PRIVATE HAVE_SECURITY=1 FASTDDS_ENFORCE_LOG_INFO=1 FASTDDS_DSOFTBUS_ROS214_PATCH=7)
target_include_directories(ros_diagnostic_stream PRIVATE ../../../integrations/ros2)
target_link_libraries(ros_diagnostic_stream PRIVATE fastrtps fastcdr)
