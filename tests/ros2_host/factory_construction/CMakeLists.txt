cmake_minimum_required(VERSION 3.24)
project(FactoryConstructionCause LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This actual factory loader fixture uses the existing Windows Host profile")
endif()
find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(factory_construction_probe SHARED ../factory_construction_probe.cpp)
target_link_libraries(factory_construction_probe PRIVATE fastrtps)
target_compile_features(factory_construction_probe PRIVATE cxx_std_17)
add_executable(factory_construction_tests ../factory_construction_tests.cpp)
target_link_libraries(factory_construction_tests PRIVATE fastrtps)
target_compile_features(factory_construction_tests PRIVATE cxx_std_17)
include(CTest)
add_test(NAME factory_autoenable_failure COMMAND factory_construction_tests
  $<TARGET_FILE:factory_construction_probe> "${CMAKE_CURRENT_BINARY_DIR}/failed-enable.conf")
set_tests_properties(factory_autoenable_failure PROPERTIES TIMEOUT 20)
add_test(NAME factory_autoenable_disabled COMMAND factory_construction_tests
  $<TARGET_FILE:factory_construction_probe> "${CMAKE_CURRENT_BINARY_DIR}/disabled-enable.conf" autoenable-disabled)
set_tests_properties(factory_autoenable_disabled PROPERTIES TIMEOUT 20)
add_executable(factory_lease_tests ../hook_close_tests.cpp)
target_compile_features(factory_lease_tests PRIVATE cxx_std_17)
target_compile_options(factory_lease_tests PRIVATE /UNDEBUG)
target_link_libraries(factory_lease_tests PRIVATE Threads::Threads)
add_test(NAME factory_lease_creation_boundary COMMAND factory_lease_tests)
set_tests_properties(factory_lease_creation_boundary PROPERTIES TIMEOUT 20)
