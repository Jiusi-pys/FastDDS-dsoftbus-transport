find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED COMPONENTS shared
    PATHS "${fastdds_DIR}" NO_DEFAULT_PATH)
if(BINDING_PRELOAD_GLOBAL)
    get_property(preloaded_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
    foreach(preloaded IN LISTS preloaded_targets)
        set_property(TARGET "${preloaded}" PROPERTY IMPORTED_GLOBAL TRUE)
    endforeach()
endif()
if(BINDING_MAP_TARGET)
    # Negative fixture only. A genuine RELEASE import stays valid while the
    # effective Release mapping selects another configuration's existing files.
    if(BINDING_MAP_TARGET STREQUAL "fastdds")
        set(alternate fastcdr)
    else()
        set(alternate fastdds)
    endif()
    get_target_property(alternate_library "${alternate}" IMPORTED_LOCATION_RELEASE)
    set_property(TARGET "${BINDING_MAP_TARGET}" APPEND PROPERTY IMPORTED_CONFIGURATIONS DEBUG)
    set_property(TARGET "${BINDING_MAP_TARGET}" PROPERTY IMPORTED_LOCATION_DEBUG "${alternate_library}")
    if(WIN32)
        get_target_property(alternate_implib "${alternate}" IMPORTED_IMPLIB_RELEASE)
        set_property(TARGET "${BINDING_MAP_TARGET}" PROPERTY IMPORTED_IMPLIB_DEBUG "${alternate_implib}")
    endif()
    set_property(TARGET "${BINDING_MAP_TARGET}" PROPERTY MAP_IMPORTED_CONFIG_RELEASE "${BINDING_RELEASE_MAPPING}")
endif()
add_subdirectory("${MDDS_CONSUMER_ROOT}" mdds)
