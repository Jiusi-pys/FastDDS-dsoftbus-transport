cmake_minimum_required(VERSION 3.22)
project(host_facade_variant_consumer LANGUAGES C CXX)
add_subdirectory("${FASTDDS_DSOFTBUS_ROOT}" project)
add_executable(facade_variant_consumer main.cpp)
if(VARIANT STREQUAL "host")
    target_link_libraries(facade_variant_consumer PRIVATE fastdds_dsoftbus_host_injected_facade)
elseif(VARIANT STREQUAL "production")
    target_link_libraries(facade_variant_consumer PRIVATE fastdds_dsoftbus_transport_facade)
elseif(VARIANT STREQUAL "both")
    target_link_libraries(facade_variant_consumer PRIVATE fastdds_dsoftbus_host_injected_facade fastdds_dsoftbus_transport_facade)
else()
    message(FATAL_ERROR "unknown variant")
endif()
