cmake_minimum_required(VERSION 3.22)
project(StandardsCBackend LANGUAGES C CXX)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
else()
    message(FATAL_ERROR "Select exact supported Fast DDS ABI")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
# Reuse B3's exact ABI and full input/generated-file provenance gate.
add_subdirectory(../cpp_backend b3)
get_target_property(b3_sources b3_contract SOURCES)
list(FILTER b3_sources EXCLUDE REGEX "backend_contract.cpp$")
add_library(standards_b4 STATIC b4.cpp ${b3_sources})
target_compile_features(standards_b4 PRIVATE cxx_std_17)
target_include_directories(standards_b4 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${B3_GENERATED_DIR}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(standards_b4 PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1 B3_HEADER="BoundedRecordPubSubTypes.h")
    target_link_libraries(standards_b4 PRIVATE fastrtps fastcdr Threads::Threads)
else()
    target_compile_definitions(standards_b4 PRIVATE B3_HEADER="BoundedRecordPubSubTypes.hpp")
    target_link_libraries(standards_b4 PRIVATE fastdds fastcdr Threads::Threads)
endif()
option(B4_DSOFTBUS "Build board adapter against installed stock transport" OFF)
if(B4_DSOFTBUS)
    find_package(FastDDSDSoftBusStock CONFIG REQUIRED)
    target_compile_definitions(standards_b4 PRIVATE B4_DSOFTBUS=1)
    target_link_libraries(standards_b4 PRIVATE FastDDSDSoftBusStock::Transport ${CMAKE_DL_LIBS})
endif()
add_executable(b4_client c_client.c)
set_target_properties(b4_client PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_link_libraries(b4_client PRIVATE standards_b4)
include(CTest)
add_test(NAME standards.b4_ownership COMMAND b4_client)
add_executable(b4_runtime_owner_test runtime_owner_test.cpp)
target_compile_features(b4_runtime_owner_test PRIVATE cxx_std_17)
target_link_libraries(b4_runtime_owner_test PRIVATE Threads::Threads)
add_test(NAME standards.b4_close_failure COMMAND b4_runtime_owner_test)
