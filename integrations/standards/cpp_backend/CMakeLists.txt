cmake_minimum_required(VERSION 3.22)
project(StandardsCppBackend LANGUAGES CXX)
set(DDS_STANDARDS_FASTDDS "" CACHE STRING "Exact Fast DDS ABI")
set(B3_GENERATED_DIR "" CACHE PATH "Hash-recorded generated artifact directory")
set(B3_IDL "" CACHE FILEPATH "Original preprocessed IDL source")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
else()
    message(FATAL_ERROR "Select exact Fast DDS 2.14.5 or 3.6.2")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
file(READ "${B3_GENERATED_DIR}/generation.json" manifest)
string(JSON status GET "${manifest}" status)
string(JSON version GET "${manifest}" fastdds)
string(JSON expected_idl GET "${manifest}" idl_sha256)
file(SHA256 "${B3_IDL}" actual_idl)
if(NOT status STREQUAL "PASS" OR NOT version STREQUAL DDS_STANDARDS_FASTDDS OR NOT expected_idl STREQUAL actual_idl)
    message(FATAL_ERROR "Generated artifact provenance/ABI/IDL mismatch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${B3_GENERATED_DIR}/generation.json" "${B3_IDL}")
foreach(input_group dependencies backend_sources)
    string(JSON input_count ERROR_VARIABLE input_error LENGTH "${manifest}" ${input_group})
    if(NOT input_error AND input_count GREATER 0)
        math(EXPR input_last "${input_count} - 1")
        foreach(input_index RANGE ${input_last})
            string(JSON input_path MEMBER "${manifest}" ${input_group} ${input_index})
            string(JSON input_expected GET "${manifest}" ${input_group} "${input_path}")
            file(SHA256 "${input_path}" input_actual)
            if(NOT input_actual STREQUAL input_expected)
                message(FATAL_ERROR "Generation input changed: ${input_path}")
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input_path}")
        endforeach()
    endif()
endforeach()
string(JSON count LENGTH "${manifest}" generated)
math(EXPR last "${count} - 1")
set(sources)
foreach(index RANGE ${last})
    string(JSON name MEMBER "${manifest}" generated ${index})
    if(NOT name MATCHES "^[A-Za-z][A-Za-z0-9_]*\\.(h|hpp|cxx|ipp)$")
        message(FATAL_ERROR "Nonportable generated artifact name")
    endif()
    file(SHA256 "${B3_GENERATED_DIR}/${name}" actual)
    string(JSON expected GET "${manifest}" generated "${name}")
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Generated file hash mismatch: ${name}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${B3_GENERATED_DIR}/${name}")
    if(name MATCHES "\\.cxx$")
        list(APPEND sources "${B3_GENERATED_DIR}/${name}")
    endif()
endforeach()
get_filename_component(stem "${B3_IDL}" NAME_WE)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    set(header "${stem}PubSubTypes.h")
else()
    set(header "${stem}PubSubTypes.hpp")
endif()
add_executable(b3_contract backend_contract.cpp ${sources})
target_compile_features(b3_contract PRIVATE cxx_std_17)
target_compile_definitions(b3_contract PRIVATE B3_HEADER="${header}")
string(JSON backend ERROR_VARIABLE backend_error GET "${manifest}" backend)
if(NOT backend_error AND backend STREQUAL "standards-b3-cpp-v1")
    target_compile_definitions(b3_contract PRIVATE B3_OWN_BACKEND=1)
endif()
if(stem STREQUAL "B3Cases")
    target_compile_definitions(b3_contract PRIVATE B3_CASES=1)
elseif(NOT stem STREQUAL "BoundedRecord")
    message(FATAL_ERROR "No acceptance harness for this fixture")
endif()
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(b3_contract PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1)
endif()
target_include_directories(b3_contract PRIVATE "${B3_GENERATED_DIR}")
target_link_libraries(b3_contract PRIVATE ${dds_target} fastcdr Threads::Threads)
include(CTest)
add_test(NAME standards.b3_contract COMMAND b3_contract)
