cmake_minimum_required(VERSION 3.22)
project(B3Combination LANGUAGES CXX)
set(DDS_STANDARDS_FASTDDS "" CACHE STRING "Exact Fast DDS ABI")
set(B3_DIRS "" CACHE STRING "Three generated directories, first/second/combined")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
    set(ext h)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
    set(ext hpp)
else()
    message(FATAL_ERROR "Select an exact supported Fast DDS release")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
list(LENGTH B3_DIRS dir_count)
if(NOT dir_count EQUAL 3)
    message(FATAL_ERROR "Combination requires exactly three generated directories")
endif()
set(sources)
foreach(directory IN LISTS B3_DIRS)
    file(READ "${directory}/generation.json" manifest)
    string(JSON status GET "${manifest}" status)
    string(JSON version GET "${manifest}" fastdds)
    if(NOT status STREQUAL "PASS" OR NOT version STREQUAL DDS_STANDARDS_FASTDDS)
        message(FATAL_ERROR "Combination artifact ABI/status mismatch")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${directory}/generation.json")
    foreach(input_group dependencies backend_sources)
        string(JSON input_count LENGTH "${manifest}" ${input_group})
        math(EXPR input_last "${input_count} - 1")
        foreach(input_index RANGE ${input_last})
            string(JSON input_path MEMBER "${manifest}" ${input_group} ${input_index})
            string(JSON input_expected GET "${manifest}" ${input_group} "${input_path}")
            file(SHA256 "${input_path}" input_actual)
            if(NOT input_actual STREQUAL input_expected)
                message(FATAL_ERROR "Combination generation input changed: ${input_path}")
            endif()
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input_path}")
        endforeach()
    endforeach()
    string(JSON count LENGTH "${manifest}" generated)
    math(EXPR last "${count} - 1")
    foreach(index RANGE ${last})
        string(JSON name MEMBER "${manifest}" generated ${index})
        if(NOT name MATCHES "^[A-Za-z][A-Za-z0-9_]*\\.(h|hpp|cxx|ipp)$")
            message(FATAL_ERROR "Nonportable artifact name")
        endif()
        string(JSON expected GET "${manifest}" generated "${name}")
        file(SHA256 "${directory}/${name}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Combination artifact hash mismatch: ${name}")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${directory}/${name}")
        if(name MATCHES "\\.cxx$")
            list(APPEND sources "${directory}/${name}")
        endif()
    endforeach()
endforeach()
list(GET B3_DIRS 0 first)
list(GET B3_DIRS 1 second)
list(GET B3_DIRS 2 combined)
add_executable(b3_combination combined_contract.cpp ${sources})
target_compile_features(b3_combination PRIVATE cxx_std_17)
target_compile_definitions(b3_combination PRIVATE
    B3_FIRST="${first}/BoundedRecordPubSubTypes.${ext}"
    B3_SECOND="${second}/B3CasesPubSubTypes.${ext}"
    B3_COMBINED="${combined}/B3CombinedPubSubTypes.${ext}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(b3_combination PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1)
endif()
target_link_libraries(b3_combination PRIVATE ${dds_target} fastcdr)
include(CTest)
add_test(NAME standards.b3_combination COMMAND b3_combination)
