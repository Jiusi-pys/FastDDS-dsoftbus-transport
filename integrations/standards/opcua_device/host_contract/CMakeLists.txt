cmake_minimum_required(VERSION 3.22)
project(D5ParticipantContract LANGUAGES CXX)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
else()
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
endif()
find_package(Threads REQUIRED)
add_executable(d5_participant_contract contract.cpp)
target_compile_features(d5_participant_contract PRIVATE cxx_std_17)
target_link_libraries(d5_participant_contract PRIVATE ${dds_target} Threads::Threads)
enable_testing()
add_test(NAME standards.d5_participant_transport COMMAND d5_participant_contract)
