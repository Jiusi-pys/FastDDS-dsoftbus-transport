cmake_minimum_required(VERSION 3.22)
project(StandardsUaDevice LANGUAGES C CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT OHOS_ARCH STREQUAL "arm64-v8a")
    message(FATAL_ERROR "D5 requires the explicit OpenHarmony arm64 toolchain")
endif()
find_package(open62541 1.4.21 EXACT CONFIG REQUIRED)
find_package(FastDDSDSoftBusStock CONFIG REQUIRED)
add_subdirectory(../c_backend b4)
get_target_property(record_sources b3_contract SOURCES)
list(FILTER record_sources EXCLUDE REGEX "backend_contract.cpp$")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    set(dds_target fastrtps)
    set(record_header BoundedRecordPubSubTypes.h)
    set(rpc_header RpcMessagePubSubTypes.h)
else()
    set(dds_target fastdds)
    set(record_header BoundedRecordPubSubTypes.hpp)
    set(rpc_header RpcMessagePubSubTypes.hpp)
endif()
# D3's Windows build guard stays intact; validate the fresh native RPC output.
file(READ "${D3_GENERATED_DIR}/generation.json" manifest)
string(JSON status GET "${manifest}" status)
string(JSON abi GET "${manifest}" fastdds)
string(JSON expected GET "${manifest}" idl_sha256)
file(SHA256 "${D3_IDL}" actual)
if(NOT status STREQUAL "PASS" OR NOT abi STREQUAL DDS_STANDARDS_FASTDDS OR NOT expected STREQUAL actual)
    message(FATAL_ERROR "D5 RPC IDL/ABI provenance mismatch")
endif()
string(JSON count LENGTH "${manifest}" generated)
math(EXPR last "${count}-1")
set(rpc_sources)
foreach(i RANGE ${last})
    string(JSON name MEMBER "${manifest}" generated ${i})
    if(NOT name MATCHES "^[A-Za-z][A-Za-z0-9_]*\\.(h|hpp|cxx|ipp)$")
        message(FATAL_ERROR "D5 RPC filename invalid")
    endif()
    string(JSON expected GET "${manifest}" generated "${name}")
    file(SHA256 "${D3_GENERATED_DIR}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "D5 RPC generated source changed")
    endif()
    if(name MATCHES "\\.cxx$")
        list(APPEND rpc_sources "${D3_GENERATED_DIR}/${name}")
    endif()
endforeach()
add_library(d5_records STATIC ${record_sources})
add_library(d5_rpc STATIC ${rpc_sources})
foreach(target d5_records d5_rpc)
    target_compile_features(${target} PUBLIC cxx_std_17)
    target_include_directories(${target} PUBLIC "${B3_GENERATED_DIR}" "${D3_GENERATED_DIR}")
    target_link_libraries(${target} PUBLIC ${dds_target} fastcdr Threads::Threads)
endforeach()
foreach(pair
    "d1_gateway|opcua_gateway/gateway.cpp" "d1_peer|opcua_gateway/plain_probe.cpp"
    "d2_gateway|dds_ua_server/server.cpp" "d2_peer|dds_ua_server/peer.cpp"
    "d3_actor|request_reply/actor.cpp" "d3_peer|request_reply/peer.cpp"
    "d4_service|opcua_complete/service_gateway.cpp" "d4_history|opcua_complete/history_gateway.cpp"
    "d4_single_peer|opcua_complete/native_peer.cpp")
    string(REPLACE "|" ";" fields "${pair}")
    list(GET fields 0 target)
    list(GET fields 1 app)
    add_executable(${target} entry.cpp)
    target_compile_definitions(${target} PRIVATE D5_APP_SOURCE="../${app}" B3_HEADER="${record_header}" D3_HEADER="${rpc_header}")
    target_include_directories(${target} PRIVATE ../request_reply)
    target_link_libraries(${target} PRIVATE d5_records d5_rpc FastDDSDSoftBusStock::Transport ${CMAKE_DL_LIBS})
    if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
        target_compile_definitions(${target} PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1 D3_FASTDDS_214=1)
    endif()
endforeach()
add_executable(d4_peer peer_mux.cpp)
target_compile_definitions(d4_peer PRIVATE B3_HEADER="${record_header}" D3_HEADER="${rpc_header}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(d4_peer PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1 D3_FASTDDS_214=1)
endif()
target_link_libraries(d4_peer PRIVATE d5_records d5_rpc FastDDSDSoftBusStock::Transport ${CMAKE_DL_LIBS})
target_link_libraries(d1_gateway PRIVATE standards_b4)
foreach(target d1_gateway d2_gateway d4_service d4_history)
    target_link_libraries(${target} PRIVATE open62541::open62541)
endforeach()
add_executable(d1_fixture ../opcua_gateway/independent_server/server.cpp)
target_compile_features(d1_fixture PRIVATE cxx_std_17)
target_link_libraries(d1_fixture PRIVATE open62541::open62541)
add_executable(d5_query ../web_gateway/c4_query.cpp)
target_include_directories(d5_query PRIVATE ../../../communication_dsoftbus/interfaces/kits/bus_center ../../../communication_dsoftbus/interfaces/kits/common)
target_link_libraries(d5_query PRIVATE FastDDSDSoftBusStock::Transport ${CMAKE_DL_LIBS})
