cmake_minimum_required(VERSION 3.22)
project(StandardsWebGateway LANGUAGES CXX)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(c1_dds fastrtps)
    set(c1_header BoundedRecordPubSubTypes.h)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(c1_dds fastdds)
    set(c1_header BoundedRecordPubSubTypes.hpp)
else()
    message(FATAL_ERROR "Select exact Fast DDS ABI")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS SSL Crypto)
find_package(LibXml2 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(C1_BOOST_INCLUDE_DIR "" CACHE PATH "Pinned Boost 1.87 headers")
file(READ "${C1_BOOST_INCLUDE_DIR}/boost/version.hpp" boost_version)
if(NOT boost_version MATCHES "#define BOOST_VERSION 108700")
    message(FATAL_ERROR "Boost 1.87.0 headers required")
endif()
add_subdirectory(../cpp_backend b3)
get_target_property(generated_sources b3_contract SOURCES)
list(FILTER generated_sources EXCLUDE REGEX "backend_contract.cpp$")
set(schema_dir "${CMAKE_CURRENT_BINARY_DIR}/schemas")
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${repo}/scripts/standards/prepare_web_schema.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/schemas" "${schema_dir}" RESULT_VARIABLE prepared)
if(NOT prepared EQUAL 0)
    message(FATAL_ERROR "Offline schema preparation failed")
endif()
file(GLOB schema_inputs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/schemas/original/*.xsd")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${schema_inputs}
    "${CMAKE_CURRENT_SOURCE_DIR}/schemas/provenance.json" "${repo}/scripts/standards/prepare_web_schema.py")
file(GLOB runtime_schemas "${schema_dir}/*.xsd")
list(LENGTH runtime_schemas schema_count)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/schema_hashes.hpp" "#pragma once\n#include <array>\n#include <utility>\n#define C1_SCHEMA_DIR R\"path(${schema_dir})path\"\nnamespace c1 { inline constexpr std::array<std::pair<const char*,const char*>,${schema_count}> schema_hashes={{\n")
foreach(path IN LISTS runtime_schemas)
    file(SHA256 "${path}" digest)
    get_filename_component(name "${path}" NAME)
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/schema_hashes.hpp" "{\"${name}\",\"${digest}\"},\n")
endforeach()
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/schema_hashes.hpp" "}}; }\n")
add_library(c1_types STATIC ${generated_sources})
target_include_directories(c1_types PUBLIC "${B3_GENERATED_DIR}")
target_link_libraries(c1_types PUBLIC ${c1_dds} fastcdr Threads::Threads)
target_compile_features(c1_types PUBLIC cxx_std_17)
target_compile_definitions(c1_types PUBLIC B3_HEADER="${c1_header}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(c1_types PUBLIC FASTDDS_DSOFTBUS_FASTDDS_214=1)
endif()
add_library(c1_core STATIC xml.cpp model.cpp)
target_include_directories(c1_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(c1_core PUBLIC c1_types LibXml2::LibXml2 OpenSSL::Crypto)
add_executable(webdds_gateway server.cpp)
target_include_directories(webdds_gateway PRIVATE "${C1_BOOST_INCLUDE_DIR}")
target_compile_definitions(webdds_gateway PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_LIB _WIN32_WINNT=0x0601)
target_link_libraries(webdds_gateway PRIVATE c1_core OpenSSL::SSL)
if(WIN32)
    target_link_libraries(webdds_gateway PRIVATE ws2_32 mswsock)
endif()
add_executable(c1_peer peer.cpp)
target_link_libraries(c1_peer PRIVATE c1_types)
