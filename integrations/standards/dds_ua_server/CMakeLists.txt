cmake_minimum_required(VERSION 3.22)
project(StandardsDdsUaServer LANGUAGES C CXX)
if(NOT WIN32 OR CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "D2 is a Windows host profile; D5 owns the device path")
endif()
find_package(open62541 1.4.21 EXACT CONFIG REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../opcua_gateway/verify_ua.cmake")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
    set(dds_header "BoundedRecordPubSubTypes.h")
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
    set(dds_header "BoundedRecordPubSubTypes.hpp")
else()
    message(FATAL_ERROR "Select exact DDS ABI")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
add_subdirectory(../cpp_backend b3)
get_target_property(type_sources b3_contract SOURCES)
list(FILTER type_sources EXCLUDE REGEX "backend_contract.cpp$")
foreach(target dds_ua_gateway d2_peer)
    if(target STREQUAL "dds_ua_gateway")
        set(main server.cpp)
    else()
        set(main peer.cpp)
    endif()
    add_executable(${target} ${main} ${type_sources})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE "${B3_GENERATED_DIR}")
    target_compile_definitions(${target} PRIVATE B3_HEADER="${dds_header}")
    if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
        target_compile_definitions(${target} PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1)
    endif()
    target_link_libraries(${target} PRIVATE ${dds_target} fastcdr Threads::Threads)
endforeach()
target_link_libraries(dds_ua_gateway PRIVATE open62541::open62541)
add_executable(d2_contract contract.cpp)
target_compile_features(d2_contract PRIVATE cxx_std_17)
target_link_libraries(d2_contract PRIVATE open62541::open62541 Threads::Threads)
add_test(NAME standards.d2_contract COMMAND d2_contract)
