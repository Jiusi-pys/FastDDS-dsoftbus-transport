cmake_minimum_required(VERSION 3.22)
project(StandardsUaToDds LANGUAGES C CXX)
if(NOT WIN32 OR CMAKE_CROSSCOMPILING OR B4_DSOFTBUS)
    message(FATAL_ERROR "D1 target is Windows host UDP; device path belongs to D5")
endif()
find_package(open62541 1.4.21 EXACT CONFIG REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/verify_ua.cmake")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
else()
    message(FATAL_ERROR "Select exact Fast DDS ABI")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
add_subdirectory(../c_backend b4)
add_executable(ua_dds_gateway gateway.cpp)
target_compile_features(ua_dds_gateway PRIVATE cxx_std_17)
target_link_libraries(ua_dds_gateway PRIVATE standards_b4 open62541::open62541 Threads::Threads)
add_executable(d1_contract contract.cpp)
target_compile_features(d1_contract PRIVATE cxx_std_17)
target_link_libraries(d1_contract PRIVATE open62541::open62541 Threads::Threads)
add_test(NAME standards.d1_contract COMMAND d1_contract)
add_test(NAME standards.d1_build_guards COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/build_guard_contract.py")
# The independent DDS probe links generated type support and public DDS only.
get_target_property(probe_sources standards_b4 SOURCES)
list(FILTER probe_sources EXCLUDE REGEX "b4.cpp$")
add_executable(d1_plain_probe plain_probe.cpp ${probe_sources})
target_compile_features(d1_plain_probe PRIVATE cxx_std_17)
target_include_directories(d1_plain_probe PRIVATE "${B3_GENERATED_DIR}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(d1_plain_probe PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1 B3_HEADER="BoundedRecordPubSubTypes.h")
    target_link_libraries(d1_plain_probe PRIVATE fastrtps fastcdr Threads::Threads)
else()
    target_compile_definitions(d1_plain_probe PRIVATE B3_HEADER="BoundedRecordPubSubTypes.hpp")
    target_link_libraries(d1_plain_probe PRIVATE fastdds fastcdr Threads::Threads)
endif()
