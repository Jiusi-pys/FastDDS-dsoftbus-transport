cmake_minimum_required(VERSION 3.22)
project(StandardsRequestReply LANGUAGES CXX)
if(NOT WIN32 OR CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "D3 is a Windows host profile; D5 owns device qualification")
endif()
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
    set(type_header RpcMessagePubSubTypes.h)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
    set(type_header RpcMessagePubSubTypes.hpp)
else()
    message(FATAL_ERROR "Select exact DDS ABI")
endif()
find_package(fastcdr 2 CONFIG REQUIRED)
find_package(Threads REQUIRED)
# SDK usage requirements contain TARGET_EXISTS expressions. When this project is
# added from a parent, those expressions must still see its imported targets.
get_property(sdk_imports DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(imported IN LISTS sdk_imports)
    set_property(TARGET ${imported} PROPERTY IMPORTED_GLOBAL TRUE)
endforeach()
file(READ "${D3_GENERATED_DIR}/generation.json" manifest)
string(JSON status GET "${manifest}" status)
string(JSON abi GET "${manifest}" fastdds)
string(JSON expected GET "${manifest}" idl_sha256)
file(SHA256 "${D3_IDL}" actual)
if(NOT status STREQUAL "PASS" OR NOT abi STREQUAL DDS_STANDARDS_FASTDDS OR NOT expected STREQUAL actual)
    message(FATAL_ERROR "D3 generation ABI/IDL mismatch")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${D3_GENERATED_DIR}/generation.json" "${D3_IDL}")
foreach(group dependencies backend_sources)
    string(JSON count LENGTH "${manifest}" ${group})
    if(count GREATER 0)
      math(EXPR last "${count}-1")
      foreach(i RANGE ${last})
        string(JSON path MEMBER "${manifest}" ${group} ${i})
        string(JSON expected GET "${manifest}" ${group} "${path}")
        file(SHA256 "${path}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "D3 generation input changed: ${path}")
        endif()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
      endforeach()
    endif()
endforeach()
string(JSON count LENGTH "${manifest}" generated)
math(EXPR last "${count}-1")
set(type_sources)
foreach(i RANGE ${last})
    string(JSON name MEMBER "${manifest}" generated ${i})
    if(NOT name MATCHES "^[A-Za-z][A-Za-z0-9_]*\\.(h|hpp|cxx|ipp)$")
        message(FATAL_ERROR "D3 invalid generated filename")
    endif()
    string(JSON expected GET "${manifest}" generated "${name}")
    file(SHA256 "${D3_GENERATED_DIR}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "D3 generated file changed")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${D3_GENERATED_DIR}/${name}")
    if(name MATCHES "\\.cxx$")
        list(APPEND type_sources "${D3_GENERATED_DIR}/${name}")
    endif()
endforeach()
add_library(d3_request_reply INTERFACE)
target_compile_features(d3_request_reply INTERFACE cxx_std_17)
target_include_directories(d3_request_reply INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}" "${D3_GENERATED_DIR}")
target_compile_definitions(d3_request_reply INTERFACE D3_HEADER="${type_header}")
target_link_libraries(d3_request_reply INTERFACE ${dds_target} fastcdr Threads::Threads)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(d3_request_reply INTERFACE D3_FASTDDS_214=1)
endif()
foreach(target d3_actor d3_contract)
    if(target STREQUAL "d3_actor")
        set(main actor.cpp)
    else()
        set(main contract.cpp)
    endif()
    add_executable(${target} ${main})
    target_link_libraries(${target} PRIVATE d3_request_reply)
endforeach()
# Independent SDK peer intentionally does not include or link the D3 library.
add_executable(d3_peer peer.cpp ${type_sources})
target_compile_features(d3_peer PRIVATE cxx_std_17)
target_include_directories(d3_peer PRIVATE "${D3_GENERATED_DIR}")
target_compile_definitions(d3_peer PRIVATE D3_HEADER="${type_header}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(d3_peer PRIVATE D3_FASTDDS_214=1)
endif()
target_link_libraries(d3_peer PRIVATE ${dds_target} fastcdr Threads::Threads)
get_target_property(sdk_library ${dds_target} IMPORTED_IMPLIB_RELEASE)
get_target_property(sdk_runtime ${dds_target} IMPORTED_LOCATION_RELEASE)
if(NOT EXISTS "${sdk_library}" OR NOT EXISTS "${sdk_runtime}")
    message(FATAL_ERROR "D3 requires resolved Release DDS import library and DLL")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/d3-dependency.json" "{\"library\":\"${sdk_library}\",\"runtime\":\"${sdk_runtime}\"}\n")
include(CTest)
add_test(NAME standards.d3_contract COMMAND d3_contract)
