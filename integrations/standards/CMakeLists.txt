cmake_minimum_required(VERSION 3.22)
project(DdsStandardsBaseline LANGUAGES CXX)

set(DDS_STANDARDS_FASTDDS "3.6.2" CACHE STRING "Exact Fast DDS release")
set_property(CACHE DDS_STANDARDS_FASTDDS PROPERTY STRINGS 2.14.5 3.6.2)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
    set(dds_target fastrtps)
elseif(DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
    find_package(fastdds 3.6.2.0 EXACT CONFIG REQUIRED)
    set(dds_target fastdds)
else()
    message(FATAL_ERROR "DDS_STANDARDS_FASTDDS must be 2.14.5 or 3.6.2")
endif()
find_package(Threads REQUIRED)
find_package(fastcdr 2 CONFIG REQUIRED)
add_executable(dds_type_probe type_probe.cpp)
target_compile_features(dds_type_probe PRIVATE cxx_std_17)
target_link_libraries(dds_type_probe PRIVATE ${dds_target} fastcdr Threads::Threads)
target_include_directories(dds_type_probe PRIVATE ../../tests/device_suite)
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(dds_type_probe PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1)
endif()
install(TARGETS dds_type_probe RUNTIME DESTINATION bin)

# Optional white-box contract tests against the actual out-of-tree A1 patch.
# The source path is explicit so stock builds never acquire patched behaviour.
set(DDS_STANDARDS_ASSIGNABILITY_SOURCE "" CACHE PATH "Patched Fast DDS 3.6.2 source tree")
if(DDS_STANDARDS_ASSIGNABILITY_SOURCE)
    if(NOT DDS_STANDARDS_FASTDDS STREQUAL "3.6.2")
        message(FATAL_ERROR "A1 assignability contract tests require Fast DDS 3.6.2")
    endif()
    add_executable(dds_assignability_contract ../../tests/standards_assignability_tests.cpp)
    target_compile_features(dds_assignability_contract PRIVATE cxx_std_17)
    target_include_directories(dds_assignability_contract PRIVATE "${DDS_STANDARDS_ASSIGNABILITY_SOURCE}/src/cpp")
    target_link_libraries(dds_assignability_contract PRIVATE ${dds_target} fastcdr)
endif()

set(DDS_STANDARDS_GENERATED_DIR "" CACHE PATH "Directory produced by scripts/standards/generate_types.py")
if(DDS_STANDARDS_GENERATED_DIR)
    file(READ "${DDS_STANDARDS_GENERATED_DIR}/generation.json" generation)
    string(JSON generated_version GET "${generation}" fastdds)
    string(JSON generated_status GET "${generation}" status)
    if(NOT generated_version STREQUAL DDS_STANDARDS_FASTDDS OR NOT generated_status STREQUAL "PASS")
        message(FATAL_ERROR "Generated type provenance must match the selected Fast DDS version")
    endif()
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/idl/BoundedRecord.idl" idl_hash)
    string(JSON generated_idl_hash GET "${generation}" idl_sha256)
    if(NOT idl_hash STREQUAL generated_idl_hash)
        message(FATAL_ERROR "IDL changed since generation")
    endif()
    if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
        set(generated_files BoundedRecord.h BoundedRecord.cxx BoundedRecordCdrAux.hpp
            BoundedRecordCdrAux.ipp BoundedRecordPubSubTypes.h BoundedRecordPubSubTypes.cxx)
        set(generated_sources BoundedRecord.cxx BoundedRecordPubSubTypes.cxx)
    else()
        set(generated_files BoundedRecord.hpp BoundedRecordCdrAux.hpp BoundedRecordCdrAux.ipp
            BoundedRecordPubSubTypes.cxx BoundedRecordPubSubTypes.hpp
            BoundedRecordTypeObjectSupport.cxx BoundedRecordTypeObjectSupport.hpp)
        set(generated_sources BoundedRecordPubSubTypes.cxx BoundedRecordTypeObjectSupport.cxx)
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        "${DDS_STANDARDS_GENERATED_DIR}/generation.json" "${CMAKE_CURRENT_SOURCE_DIR}/idl/BoundedRecord.idl")
    foreach(name IN LISTS generated_files)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DDS_STANDARDS_GENERATED_DIR}/${name}")
        file(SHA256 "${DDS_STANDARDS_GENERATED_DIR}/${name}" actual_hash)
        string(JSON expected_hash GET "${generation}" generated "${name}")
        if(NOT actual_hash STREQUAL expected_hash)
            message(FATAL_ERROR "Generated file hash mismatch: ${name}")
        endif()
    endforeach()
    list(TRANSFORM generated_sources PREPEND "${DDS_STANDARDS_GENERATED_DIR}/")
    add_executable(dds_generated_roundtrip generated_roundtrip.cpp ${generated_sources})
    if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
        target_compile_definitions(dds_generated_roundtrip PRIVATE FASTDDS_DSOFTBUS_FASTDDS_214=1)
    endif()
    target_compile_features(dds_generated_roundtrip PRIVATE cxx_std_17)
    target_include_directories(dds_generated_roundtrip PRIVATE "${DDS_STANDARDS_GENERATED_DIR}")
    target_link_libraries(dds_generated_roundtrip PRIVATE ${dds_target} fastcdr)
    include(CTest)
    add_test(NAME standards.generated_roundtrip COMMAND dds_generated_roundtrip)
endif()
