cmake_minimum_required(VERSION 3.22)
project(StandardsUaComplete LANGUAGES C CXX)
if(NOT WIN32 OR CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "D4 is a Windows host profile; D5 owns device qualification")
endif()
find_package(open62541 1.4.21 EXACT CONFIG REQUIRED)
include("${CMAKE_CURRENT_SOURCE_DIR}/../opcua_gateway/verify_ua.cmake")
include(CTest)
add_subdirectory(../request_reply d3)
add_subdirectory(../cpp_backend b3)
get_target_property(record_sources b3_contract SOURCES)
list(FILTER record_sources EXCLUDE REGEX "backend_contract.cpp$")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    set(record_header BoundedRecordPubSubTypes.h)
    set(rpc_header RpcMessagePubSubTypes.h)
    set(dds_target fastrtps)
else()
    set(record_header BoundedRecordPubSubTypes.hpp)
    set(rpc_header RpcMessagePubSubTypes.hpp)
    set(dds_target fastdds)
endif()
add_executable(d4_service_gateway service_gateway.cpp)
target_link_libraries(d4_service_gateway PRIVATE d3_request_reply open62541::open62541)
foreach(target d4_history_gateway d4_contract)
    if(target STREQUAL d4_history_gateway)
        set(main history_gateway.cpp)
    else()
        set(main contract.cpp)
    endif()
    add_executable(${target} ${main} ${record_sources})
    target_include_directories(${target} PRIVATE "${B3_GENERATED_DIR}")
    target_compile_definitions(${target} PRIVATE B3_HEADER="${record_header}")
    target_link_libraries(${target} PRIVATE d3_request_reply open62541::open62541)
endforeach()
# Plain public-DDS peer: no D3/D4 implementation headers or UA libraries.
add_executable(d4_native_peer native_peer.cpp ${record_sources})
target_compile_features(d4_native_peer PRIVATE cxx_std_17)
target_include_directories(d4_native_peer PRIVATE "${D3_GENERATED_DIR}" "${B3_GENERATED_DIR}")
target_compile_definitions(d4_native_peer PRIVATE D3_HEADER="${rpc_header}" B3_HEADER="${record_header}")
if(DDS_STANDARDS_FASTDDS STREQUAL "2.14.5")
    target_compile_definitions(d4_native_peer PRIVATE D3_FASTDDS_214=1)
endif()
target_link_libraries(d4_native_peer PRIVATE ${dds_target} fastcdr Threads::Threads)
foreach(target d4_service_gateway d4_history_gateway d4_native_peer d4_contract)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endforeach()
add_test(NAME standards.d4_contract COMMAND d4_contract)
