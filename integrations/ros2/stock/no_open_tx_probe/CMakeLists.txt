cmake_minimum_required(VERSION 3.22)
project(Stock214WholeRuntimeNoOpenTxProbe LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING OR NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This immutable-archive diagnostic is Windows x64 MSVC Host only")
endif()
if(NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "19.44.35228.0")
    message(FATAL_ERROR "The reviewed archive requires the recorded MSVC compiler 19.44.35228.0")
endif()
find_package(fastrtps 2.14.5 EXACT CONFIG REQUIRED)
find_package(fastcdr 2.2.5 EXACT CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(REUSED_HOST_ARCHIVE "" CACHE FILEPATH "Read-only qualified stock214 Host archive")
set(REUSE_PROVENANCE "" CACHE FILEPATH "Reviewed immutable-archive provenance JSON")
if(NOT IS_ABSOLUTE "${REUSED_HOST_ARCHIVE}" OR NOT EXISTS "${REUSED_HOST_ARCHIVE}"
        OR NOT IS_ABSOLUTE "${REUSE_PROVENANCE}" OR NOT EXISTS "${REUSE_PROVENANCE}")
    message(FATAL_ERROR "Explicit existing archive and reviewed provenance required")
endif()
file(SHA256 "${REUSED_HOST_ARCHIVE}" _archive_sha)
file(SHA256 "${REUSE_PROVENANCE}" _provenance_sha)
if(NOT _archive_sha STREQUAL "dd08020897e3ef3e58552d8b73e96f19ecab7eafa5dba6bb462d7366a3b3d7e6"
        OR NOT _provenance_sha STREQUAL "42412c4f8d2d1bc415a1b21fc0dbd8f322a3f07ce8282b2b19fd1bc6cc43a3b9")
    message(FATAL_ERROR "Immutable archive/provenance hash differs from reviewed inputs")
endif()
file(READ "${REUSE_PROVENANCE}" _provenance)
string(JSON _recorded_path GET "${_provenance}" library path)
file(TO_CMAKE_PATH "${_recorded_path}" _recorded_path)
file(TO_CMAKE_PATH "${REUSED_HOST_ARCHIVE}" _actual_path)
if(NOT _recorded_path STREQUAL _actual_path)
    message(FATAL_ERROR "Archive path must match the reviewed provenance")
endif()
get_filename_component(_repo "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
add_library(stock_214_no_open_runtime_reuse STATIC IMPORTED)
set_target_properties(stock_214_no_open_runtime_reuse PROPERTIES IMPORTED_LOCATION "${REUSED_HOST_ARCHIVE}")
target_compile_definitions(stock_214_no_open_runtime_reuse INTERFACE
    FASTDDS_DSOFTBUS_FASTDDS_214=1 FASTDDS_DSOFTBUS_PUBLIC_SOCKET_DDS_EXPERIMENT=1
    FASTDDS_DSOFTBUS_PUBLIC_SOCKET_HOST_C_SEAM=1)
target_include_directories(stock_214_no_open_runtime_reuse INTERFACE "${_repo}/include" "${_repo}/tests")
target_link_libraries(stock_214_no_open_runtime_reuse INTERFACE fastrtps fastcdr Threads::Threads)
add_executable(stock_214_whole_runtime_no_open_tx "${_repo}/tests/ros2_host/stock_transport/whole_runtime_no_open_tx.cpp")
target_compile_features(stock_214_whole_runtime_no_open_tx PRIVATE cxx_std_17)
set_target_properties(stock_214_whole_runtime_no_open_tx PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_definitions(stock_214_whole_runtime_no_open_tx PRIVATE FASTDDS_DSOFTBUS_PUBLIC_SOCKET_RUNTIME_PROBE=1)
target_link_libraries(stock_214_whole_runtime_no_open_tx PRIVATE stock_214_no_open_runtime_reuse)
include(CTest)
if(NOT BUILD_TESTING)
    message(FATAL_ERROR "The standalone oracle requires BUILD_TESTING=ON")
endif()
add_test(NAME stock_transport.domain.whole-runtime-no-open-tx COMMAND stock_214_whole_runtime_no_open_tx)
set_tests_properties(stock_transport.domain.whole-runtime-no-open-tx PROPERTIES TIMEOUT 25)
