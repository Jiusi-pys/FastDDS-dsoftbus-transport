cmake_minimum_required(VERSION 3.22)
project(FastDDSDSoftBusStockInstalledConsumer LANGUAGES CXX)
find_package(FastDDSDSoftBusStock CONFIG REQUIRED)
add_executable(stock_transport_installed_consumer link_check.cpp)
target_link_libraries(stock_transport_installed_consumer PRIVATE FastDDSDSoftBusStock::Transport)
target_link_options(stock_transport_installed_consumer PRIVATE
    "LINKER:-z,defs" "LINKER:--no-undefined" "LINKER:--no-allow-shlib-undefined")
set(FASTDDS_DSOFTBUS_CONSUMER_LINK_DIRECTORIES "" CACHE STRING "Explicit real dependency directories; build-only rpath-link")
foreach(_provider_dir IN LISTS FASTDDS_DSOFTBUS_CONSUMER_LINK_DIRECTORIES)
    if(NOT IS_ABSOLUTE "${_provider_dir}" OR NOT IS_DIRECTORY "${_provider_dir}")
        message(FATAL_ERROR "Invalid consumer rpath-link directory: ${_provider_dir}")
    endif()
    target_link_options(stock_transport_installed_consumer PRIVATE "LINKER:-rpath-link,${_provider_dir}")
endforeach()
