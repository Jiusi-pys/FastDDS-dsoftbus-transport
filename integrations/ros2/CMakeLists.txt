    # REAL_PATH can retain Windows junctions. Reject aliases at every ancestor,
    # including when the install leaf does not exist yet. Check dependency roots
    # too, so an aliased dependency cannot evade the containment check below.
    foreach(_profile_path IN ITEMS "${CMAKE_INSTALL_PREFIX}"
            "${FASTDDS_DSOFTBUS_HISTORICAL_FASTDDS_PREFIX}" "${FASTDDS_DSOFTBUS_HISTORICAL_FASTCDR_PREFIX}")
        get_filename_component(_profile_cursor "${_profile_path}" ABSOLUTE)
        while(TRUE)
            if(IS_SYMLINK "${_profile_cursor}")
                message(FATAL_ERROR "Historical Host profile refuses symlink or junction prefix ancestry: ${_profile_cursor}")
            endif()
            if(EXISTS "${_profile_cursor}" AND NOT IS_DIRECTORY "${_profile_cursor}")
                message(FATAL_ERROR "Historical Host profile prefix ancestry must contain directories: ${_profile_cursor}")
            endif()
            cmake_path(GET _profile_cursor PARENT_PATH _profile_parent)
            if(_profile_parent STREQUAL _profile_cursor OR _profile_parent STREQUAL "")
                break()
            endif()
            set(_profile_cursor "${_profile_parent}")
        endwhile()
    endforeach()
    if(EXISTS "${CMAKE_INSTALL_PREFIX}/bin/fastdds_dsoftbus_ros2_host_bridge.dll")
        set(_installed_profile "${CMAKE_INSTALL_PREFIX}/lib/cmake/FastDDSDSoftBusTransportRos2Host/FastDDSDSoftBusTransportRos2HostConfig.cmake")
        if(NOT EXISTS "${_installed_profile}")
            message(FATAL_ERROR "Historical Host profile refuses an unclassified existing bridge install")
        endif()
        file(READ "${_installed_profile}" _installed_config)
        if(NOT _installed_config MATCHES "set\\(FastDDSDSoftBusTransportRos2Host_PROFILE \"historical-2147-security-statistics\"\\)")
            message(FATAL_ERROR "Historical Host profile refuses to overwrite another profile install")
        endif()
    endif()
