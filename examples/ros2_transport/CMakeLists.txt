cmake_minimum_required(VERSION 3.22)
project(ordinary_ros2_transport_consumer LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(example_interfaces REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  set(_transport_package FastDDSDSoftBusTransportRos2Target)
else()
  set(_transport_package FastDDSDSoftBusTransportRos2Host)
endif()
find_package(${_transport_package} CONFIG REQUIRED)
add_executable(ordinary_ros2_transport_consumer main.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  # Installed startup selects platform identity before the unchanged ROS entry.
  # The business translation unit still contains only ordinary ROS APIs.
  set_source_files_properties(main.cpp PROPERTIES COMPILE_DEFINITIONS "main=FastddsDsoftbusRosBusinessMain")
endif()
ament_target_dependencies(ordinary_ros2_transport_consumer rclcpp std_msgs example_interfaces)
target_compile_features(ordinary_ros2_transport_consumer PRIVATE cxx_std_17)
target_link_libraries(ordinary_ros2_transport_consumer ${_transport_package}::factory_configuration)
