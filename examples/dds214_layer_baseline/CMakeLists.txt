cmake_minimum_required(VERSION 3.22)
project(fastdds_dsoftbus_dds214_layer_baseline LANGUAGES CXX)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    set(_package FastDDSDSoftBusTransportRos2Target)
else()
    set(_package FastDDSDSoftBusTransportRos2Host)
endif()
find_package(${_package} CONFIG REQUIRED)
find_package(fastcdr 2 CONFIG REQUIRED)
add_executable(fastdds_dsoftbus_dds214_baseline main.cpp)
target_compile_features(fastdds_dsoftbus_dds214_baseline PRIVATE cxx_std_17)
target_link_libraries(fastdds_dsoftbus_dds214_baseline PRIVATE ${_package}::factory_configuration fastcdr)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    # Reuse installed identity-before-thread entry, independently of ROS/RMW.
    set_source_files_properties(main.cpp PROPERTIES COMPILE_DEFINITIONS "main=FastddsDsoftbusRosBusinessMain")
endif()
set_target_properties(fastdds_dsoftbus_dds214_baseline PROPERTIES
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(dds214_layer_sample_tests sample_tests.cpp)
    target_compile_features(dds214_layer_sample_tests PRIVATE cxx_std_17)
    target_link_libraries(dds214_layer_sample_tests PRIVATE fastrtps fastcdr)
    add_test(NAME dds214.sample_roundtrip_and_bounds COMMAND dds214_layer_sample_tests)
endif()
