cmake_minimum_required(VERSION 3.22)
project(ros2_action_acceptance LANGUAGES CXX)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  set(_transport_package FastDDSDSoftBusTransportRos2Target)
elseif(WIN32)
  set(_transport_package FastDDSDSoftBusTransportRos2Host)
else()
  message(FATAL_ERROR "This acceptance target requires Windows Host or OHOS")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(${_transport_package} CONFIG REQUIRED)
add_executable(ros2_action_acceptance main.cpp)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  set_source_files_properties(main.cpp PROPERTIES COMPILE_DEFINITIONS "main=FastddsDsoftbusRosBusinessMain")
endif()
ament_target_dependencies(ros2_action_acceptance rclcpp rclcpp_action example_interfaces)
target_compile_features(ros2_action_acceptance PRIVATE cxx_std_17)
target_link_libraries(ros2_action_acceptance ${_transport_package}::factory_configuration)
