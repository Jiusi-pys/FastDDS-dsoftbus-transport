cmake_minimum_required(VERSION 3.22)
project(ros2_action_acceptance LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This acceptance target is scoped to the Windows Host seam")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(FastDDSDSoftBusTransportRos2Host CONFIG REQUIRED)
add_executable(ros2_action_acceptance main.cpp)
ament_target_dependencies(ros2_action_acceptance rclcpp rclcpp_action example_interfaces)
target_compile_features(ros2_action_acceptance PRIVATE cxx_std_17)
target_link_libraries(ros2_action_acceptance FastDDSDSoftBusTransportRos2Host::factory_configuration)
