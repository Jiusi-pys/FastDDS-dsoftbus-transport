cmake_minimum_required(VERSION 3.22)
project(ros2_serialized_acceptance LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This acceptance target is scoped to the Windows Host seam")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(FastDDSDSoftBusTransportRos2Host CONFIG REQUIRED)
if(NOT FastDDSDSoftBusTransportRos2Host_PROFILE STREQUAL "default-2145")
  message(FATAL_ERROR "Serialized acceptance requires default-2145")
endif()
add_executable(ros2_serialized_acceptance main.cpp)
ament_target_dependencies(ros2_serialized_acceptance rclcpp std_msgs)
target_compile_features(ros2_serialized_acceptance PRIVATE cxx_std_17)
target_link_libraries(ros2_serialized_acceptance FastDDSDSoftBusTransportRos2Host::factory_configuration)
