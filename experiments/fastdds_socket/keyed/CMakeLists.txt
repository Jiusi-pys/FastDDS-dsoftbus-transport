cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(keyed_contract_tests LANGUAGES CXX)
    include(CTest)
    option(KEYED_VALIDATOR_ONLY "Build only portable validator tests" ON)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(keyed_validator_tests validator_tests.cpp)
    target_compile_features(keyed_validator_tests PRIVATE cxx_std_17)
    add_test(NAME experiment.keyed_validator COMMAND keyed_validator_tests)
endif()
if(KEYED_VALIDATOR_ONLY)
    return()
endif()
add_subdirectory(type)
add_library(experiment_keyed_fixture STATIC PortableFixture.cpp)
target_compile_features(experiment_keyed_fixture PUBLIC cxx_std_17)
target_include_directories(experiment_keyed_fixture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(experiment_keyed_fixture PUBLIC experimental_dsoftbus_transport experiment_keyed_type)
if(NOT CMAKE_CROSSCOMPILING)
    if(BUILD_TESTING)
        find_program(KEYED_NODE_EXECUTABLE NAMES node REQUIRED)
        target_compile_definitions(experiment_keyed_fixture PUBLIC MDDS_KEYED_HOST_CONTRACT_TESTS=1)
        add_executable(keyed_fixture_contract_tests fixture_contract_tests.cpp)
        target_link_libraries(keyed_fixture_contract_tests PRIVATE experiment_keyed_fixture)
        foreach(_contract IN ITEMS loan-read-format loan-take-format loan-read-log loan-take-log loan-read-payload loan-take-payload state-no-payload graph-initial graph-late graph-final)
            add_test(NAME experiment.keyed_fixture_${_contract} COMMAND "${KEYED_NODE_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/contract_harness.cjs" "$<TARGET_FILE:keyed_fixture_contract_tests>" "${_contract}")
            set_tests_properties(experiment.keyed_fixture_${_contract} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
        endforeach()
        if(WIN32)
            add_custom_command(TARGET keyed_fixture_contract_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_FILE:fastdds>" "$<TARGET_FILE:fastcdr>" "$<TARGET_FILE:foonathan_memory>" "$<TARGET_FILE_DIR:keyed_fixture_contract_tests>")
        endif()
    endif()
    add_executable(keyed_reference reference_main.cpp)
    target_link_libraries(keyed_reference PRIVATE experiment_keyed_fixture)
    if(BUILD_TESTING)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        foreach(_keyed_case IN ITEMS K1 K2 K3 K4_TRUE K4_FALSE H1 H2 H3 R1 R2 R3)
            add_test(NAME experiment.keyed_reference_${_keyed_case} COMMAND "${KEYED_NODE_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/reference_harness.cjs" "$<TARGET_FILE:keyed_reference>"
                "${_keyed_case}" "--root=${CMAKE_BINARY_DIR}/keyed-reference"
                "${PROJECT_SOURCE_DIR}/tests/keyed_trace_fixtures.json" "${Python3_EXECUTABLE}")
            set_tests_properties(experiment.keyed_reference_${_keyed_case} PROPERTIES TIMEOUT 225 RUN_SERIAL TRUE)
        endforeach()
    endif()
    if(WIN32)
        add_custom_command(TARGET keyed_reference POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:fastdds>" "$<TARGET_FILE:fastcdr>" "$<TARGET_FILE:foonathan_memory>" "$<TARGET_FILE_DIR:keyed_reference>")
    endif()
endif()
